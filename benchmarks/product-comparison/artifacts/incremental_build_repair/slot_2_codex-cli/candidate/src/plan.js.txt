import { diffSnapshots } from './changes.js';
import { affectedModules } from './dependents.js';
export function planBuild(graph, previous, current) {
  const delta = diffSnapshots(previous,current);
  const affected = affectedModules(graph, [...delta.changed, ...delta.removed]);
  const currentNames = new Set(Object.keys(current));
  return {
    ...delta,
    build: affected.filter(name => currentNames.has(name)),
  };
}
