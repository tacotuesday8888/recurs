function isIntegerInRange(value, minimum, maximum) {
  return Number.isInteger(value) && value >= minimum && value <= maximum;
}

export function deliveryCost(totalGrams, merchandiseCents, zone) {
  if (!isIntegerInRange(totalGrams, 0, 10000000000) ||
      !isIntegerInRange(merchandiseCents, 0, 100000000000) ||
      (zone !== 'local' && zone !== 'remote')) {
    throw new TypeError('Invalid delivery input');
  }

  if (totalGrams === 0 || (zone === 'local' && merchandiseCents >= 5000)) {
    return 0;
  }

  const kilograms = Math.ceil(totalGrams / 1000);
  return zone === 'local'
    ? 500 + 200 * kilograms
    : 900 + 350 * kilograms;
}
