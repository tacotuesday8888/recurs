const isIntegerInRange = (value, minimum, maximum) =>
  Number.isInteger(value) && value >= minimum && value <= maximum;

const isRecord = (value) =>
  value !== null && typeof value === 'object' && !Array.isArray(value);

export function normalizeCart(items, catalog) {
  if (!Array.isArray(items) || items.length > 100 ||
      !isRecord(catalog)) {
    throw new TypeError('Invalid cart input');
  }

  const quantities = new Map();

  for (const item of items) {
    if (!isRecord(item) || typeof item.sku !== 'string' ||
        item.sku.length === 0 || !isIntegerInRange(item.quantity, 1, 1000) ||
        !Object.hasOwn(catalog, item.sku)) {
      throw new TypeError('Invalid cart item');
    }

    quantities.set(item.sku, (quantities.get(item.sku) ?? 0) + item.quantity);
  }

  const lines = [];
  for (const [sku, quantity] of quantities) {
    const entry = catalog[sku];
    if (!isRecord(entry) ||
        !isIntegerInRange(entry.unitCents, 0, 1_000_000) ||
        !isIntegerInRange(entry.grams, 1, 100_000) ||
        !isIntegerInRange(entry.stock, 0, 1_000_000)) {
      throw new TypeError('Invalid catalog entry');
    }
    if (quantity > entry.stock) {
      throw new RangeError('Insufficient stock');
    }
    lines.push({ sku, quantity, unitCents: entry.unitCents, grams: entry.grams });
  }

  return lines;
}
