export function deliveryCost(totalGrams, merchandiseCents, zone) {
  if (!Number.isInteger(totalGrams) || totalGrams < 0 || totalGrams > 10000000000 ||
      !Number.isInteger(merchandiseCents) || merchandiseCents < 0 || merchandiseCents > 100000000000 ||
      (zone !== 'local' && zone !== 'remote')) {
    throw new TypeError('invalid delivery arguments');
  }
  if (totalGrams === 0) return 0;

  const kilograms = Math.ceil(totalGrams / 1000);
  if (zone === 'local' && merchandiseCents >= 5000) return 0;
  return zone === 'local' ? 500 + 200 * kilograms : 900 + 350 * kilograms;
}
