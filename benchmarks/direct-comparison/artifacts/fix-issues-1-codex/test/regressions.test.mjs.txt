import assert from 'node:assert/strict';
import test from 'node:test';
import { createService } from '../src/service.mjs';

test('closing an issue cannot change another tenant\'s issue', () => {
  const service = createService([
    { id: 'a', tenant: 'red', title: 'Red issue', status: 'open', secret: 'red secret' },
    { id: 'b', tenant: 'blue', title: 'Blue issue', status: 'open', secret: 'blue secret' },
  ]);

  assert.throws(() => service.close('red', 'b'), { code: 'NOT_FOUND' });
  assert.equal(service.get('blue', 'b').status, 'open');
});

test('pagination treats the cursor as exclusive', () => {
  const service = createService([
    { id: 'a', tenant: 'red', title: 'A', status: 'open', secret: 'a secret' },
    { id: 'b', tenant: 'red', title: 'B', status: 'open', secret: 'b secret' },
    { id: 'c', tenant: 'red', title: 'C', status: 'open', secret: 'c secret' },
  ]);

  const first = service.list('red', { limit: 2 });
  const second = service.list('red', { after: first.nextCursor, limit: 2 });
  assert.deepEqual(second.items.map(({ id }) => id), ['c']);
  assert.equal(second.nextCursor, null);
});

test('responses contain only public issue fields', () => {
  const service = createService([
    { id: 'a', tenant: 'red', title: 'A', status: 'open', secret: 'private' },
  ]);

  assert.deepEqual(service.list('red').items[0], { id: 'a', title: 'A', status: 'open' });
  assert.deepEqual(service.get('red', 'a'), { id: 'a', title: 'A', status: 'open' });
  assert.deepEqual(service.close('red', 'a'), { id: 'a', title: 'A', status: 'closed' });
});
